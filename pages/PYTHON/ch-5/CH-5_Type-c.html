Q1. Write a program that reads a text file and creates another file that is identical except that every sequence of consecutive blank spaces is replaced by a single space. 

# Function to reduce multiple spaces to single space in a file
def normalize_spaces(input_file, output_file):
    with open(input_file, 'r') as infile, open(output_file, 'w') as outfile:
        for line in infile:
            # ' '.join(line.split()) removes extra spaces between words
            normalized_line = ' '.join(line.split())
            outfile.write(normalized_line + '\n')

# Example usage
# Suppose 'input.txt' contains:
# Hello     World   !
# Python     is    fun

normalize_spaces('input.txt', 'output.txt')

# Content of 'output.txt' after running:
# Hello World !
# Python is fun
Explanation:
  • line.split() splits the line into words, ignoring extra spaces.
  • ' '.join(...) joins them back with a single space.
  • The cleaned lines are written to output.txt.
Q2. A file sports.dat contains information in the format: Event : Participant
Write a function that would read contents from sports.dat and create a file Athletic.dat copying only those records where the event name is “Athletics”. 

# Function to filter only "Athletics" records
def filter_athletics_records(input_file, output_file):
    with open(input_file, 'r') as infile, open(output_file, 'w') as outfile:
        for line in infile:
            parts = line.strip().split(":", 1)
            if len(parts) == 2:
                event = parts[0].strip()
                if event.lower() == "athletics":
                    outfile.write(line)

# Example usage
# Suppose 'sports.dat' contains:
# Athletics : Asha
# Swimming : Ravi
# Athletics : John

filter_athletics_records('sports.dat', 'Athletic.dat')

# Content of 'Athletic.dat' after running:
# Athletics : Asha
# Athletics : John
Explanation:
  • Each line is split at the first ":" to extract the event name.
  • Only lines where the event is exactly "Athletics" (case-insensitive) are written to the new file.
  • This filters the file based on the required condition.
Q3. A file contains a list of telephone numbers in the following form:

The names contain only one word, and the names and telephone numbers are separated by white spaces.
Write a program to read the file and display its contents in two columns. 

# Function to display name and phone number in two columns
def display_phonebook(file_name):
    with open(file_name, 'r') as f:
        print(f"{'Name':<15}Phone Number")
        print("-" * 30)
        for line in f:
            parts = line.strip().split()
            if len(parts) == 2:
                name, phone = parts
                print(f"{name:<15}{phone}")

# Example usage
# Suppose 'phones.txt' contains:
# Arvind 7258031
# Sachin 7259197

display_phonebook('phones.txt')

# Output:
# Name           Phone Number
# ------------------------------
# Arvind         7258031
# Sachin         7259197
Explanation:
  • line.strip().split() separates each line into name and number.
  • {name:<15} left-aligns the name in a 15-character column.
  • The result is a neat two-column display of names and phone numbers.

# Function to count "to" and "the" in a file
def count_to_the(filename):
    with open(filename, 'r') as f:
        text = f.read().lower().split()
        count_to = text.count("to")
        count_the = text.count("the")
        print(f'"to" count:', count_to)
        print(f'"the" count:', count_the)

# Output:

# "the" count: 2
Explanation:
  • File is read and converted to lowercase to ensure case-insensitive match.
  • split() separates the text into words.
  • count() is used to count exact matches of "to" and "the".

# Function to count uppercase letters
def count_uppercase(filename):
    with open(filename, 'r') as f:
        data = f.read()

def store_by_type():
    print("Enter characters (type '#' to stop):")

    with open("LOWER.txt", "w") as lower, \
         open("UPPER.txt", "w") as upper, \
         open("OTHERS.txt", "w") as others:

Q13. Write a program that counts the number of characters up to the first $ in a text file. 

def count_until_dollar(filename):
    count = 0
    with open(filename, 'r') as file:
        while True:
            char = file.read(1)
            if not char or char == '$':
                break
            count += 1
    print("Characters before first $:", count)

# Example:
# If file content: "Hello world $ this is text"
# Output: Characters before first $: 12
count_until_dollar("sample.txt")
Explanation:
  • The function reads the file character-by-character using read(1).
  • It stops counting once '$' is encountered or end-of-file is reached.
  • Useful for partial reads or stream-parsing applications.

import pickle

# (i) Function to add book records
def CreateFile():
    with open("Book.dat", "ab") as f:
        while True:
            BookNo = int(input("Enter Book Number: "))
            Book_Name = input("Enter Book Name: ")
            Author = input("Enter Author Name: ")
            Price = float(input("Enter Book Price: "))
            book = [BookNo, Book_Name, Author, Price]
            pickle.dump(book, f)
            ch = input("Add another record? (y/n): ")
            if ch.lower() != 'y':
                break

# (ii) Function to count records by author name
def CountRec(author_name):
    count = 0
    try:
        with open("Book.dat", "rb") as f:
            while True:
                book = pickle.load(f)
                if book[2].lower() == author_name.lower():
                    count += 1
    except EOFError:
        pass
    except FileNotFoundError:
        print("File not found.")
    return count

# Example usage

  • CountRec(author_name):
    • Reads all records using pickle.load()
    • Compares book[2] (Author) with the given name (case-insensitive)
    • Counts and returns the total matches

Q23. Write a function that reads a csv file and creates another csv file with the same content, but with a different delimiter. 

import csv

def convert_delimiter(source, target, old_delim=',', new_delim='|'):
    with open(source, newline='') as infile, open(target, 'w', newline='') as outfile:
        reader = csv.reader(infile, delimiter=old_delim)
        writer = csv.writer(outfile, delimiter=new_delim)
        for row in reader:
            writer.writerow(row)

Q24. Write a function that reads a csv file and creates another csv file with the same content except the lines beginning with ‘check’. 

import csv

def skip_check_lines(source, target):
    with open(source, newline='') as infile, open(target, 'w', newline='') as outfile:
        reader = csv.reader(infile)
        writer = csv.writer(outfile)
        for row in reader:
            if not row[0].lower().startswith('check'):
                writer.writerow(row)
Note: This skips any row where first column starts with the word "check".
Q25. Give any one point of difference between a binary file and a CSV file.
