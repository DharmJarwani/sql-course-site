def isEmpty(stk):
if stk == []:
return True
else:
return False
def Push(stk, item):
stk.append(item)
stackTop = len(stk) - 1
def Pop(stk):
if isEmpty(stk):
return "Underflow"
else:
item = stk.pop()
stackTop = len(stk)-1 if len(stk)>0 else None
return item
def Peek(stk):
if isEmpty(stk):
return "Underflow"
else:
return stk[len(stk)-1]
def Display(stk):
if isEmpty(stk):
print("Stack empty")
else:
print(stk[len(stk)-1], "<-- top")
for i in range(len(stk)-2, -1, -1):
print(stk[i])
