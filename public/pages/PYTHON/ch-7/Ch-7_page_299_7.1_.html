Q. Write a Python program to implement Stack operations (Push, Pop, Peek, Display). 

#################### STACK IMPLEMENTATION ####################

# Stack: implemented as a Python list
# top  : stores the index of the topmost element in the stack

# Checks whether the stack is empty
# Returns True if empty, otherwise False
def isEmpty(stk):
    if stk == []:
        return True
    else:
        return False


# Pushes (inserts) an item into the stack
# Also updates the top position after insertion
def Push(stk, item):
    stk.append(item)
    top = len(stk) - 1


# Pops (removes) the topmost element from the stack
# Returns "Underflow" if stack is empty
def Pop(stk):
    if isEmpty(stk):
        return "Underflow"
    else:
        item = stk.pop()
        if len(stk) == 0:
            top = None
        else:
            top = len(stk) - 1
        return item


# Returns the topmost element without deleting it
# If stack is empty, returns "Underflow"
def Peek(stk):
    if isEmpty(stk):
        return "Underflow"
    else:
        top = len(stk) - 1
        return stk[top]


# Displays all elements of the stack from top to bottom
# Shows "Stack empty" if there are no elements
def Display(stk):
    if isEmpty(stk):
        print("Stack empty")
    else:
        top = len(stk) - 1
        print(stk[top], "<-- top")
        for i in range(top - 1, -1, -1):
            print(stk[i])


#################### MAIN PROGRAM ####################

Stack = []      # initially stack is empty
top = None

while True:
    print("\nSTACK OPERATIONS")
    print("1. Push")
    print("2. Pop")
    print("3. Peek")
    print("4. Display stack")
    print("5. Exit")

    ch = int(input("Enter your choice (1-5) : "))

    if ch == 1:
        item = int(input("Enter item : "))
        Push(Stack, item)

    elif ch == 2:
        item = Pop(Stack)
        if item == "Underflow":
            print("Underflow! Stack is empty!")
        else:
            print("Popped item is", item)

    elif ch == 3:
        item = Peek(Stack)
        if item == "Underflow":
            print("Underflow! Stack is empty!")
        else:
            print("Topmost item is", item)

    elif ch == 4:
        Display(Stack)

    elif ch == 5:
        break

    else:
        print("Invalid choice!")
Explanation:
  • Push() inserts an element into the stack.
  • Pop() removes and returns the topmost element.
  • Peek() returns the top element without removing it.
  • Display() prints stack elements from top to bottom.
  • Program runs in a loop and performs operations based on user choice.
